Chess Assistant Pro — Stockfish Elo 400–1319 source reference

Extracted from the checked-in Tweak.xm at commit 937f863. This file preserves the exact source text and original line numbers for the low Elo policy and its shared helpers.

LIVE 400–1319 MOVE PATH: HEParamsFor, HECoreConfigure, HEParseInfo, HEMath::losses, HEMath::choice, HEMath::targetError, HEMath::fitLaw, HEMath::tilt, HEHumanBias, HECoreChooseMove, and the lowElo branch in CAPExternalEngineMove.

The HEMath namespace also contains equations used only for reporting/diagnostics or other optional timing/statistics features. Their presence in the source does not mean they all change the selected move. The selector uses only Stockfish MultiPV candidates (up to 43), not every legal move.

1608:     if (!s || !s[0] || !s[1]) return -1;
1609:     int f = s[0] - 'a';
1610:     int r = s[1] - '1';

1616: static int HEParseFEN(const char *fen, HEBoard *out) {
1617:     if (!fen || !out) return 0;
1618:     memset(out->sq, '.', sizeof(out->sq));
1619:     out->whiteToMove = 1;
1620:     out->epSquare         = -1;
1621: 
1622: 
1623:     int rank = 7, file = 0;
1624:     const char *p = fen;
1625: 
1626: 
1627:     for (; *p && *p != ' '; p++) {
1628:           if (*p == '/') { rank--; file = 0; if (rank < 0) return 0; continue; }
1629:           if (isdigit((unsigned char)*p)) { file += *p - '0'; continue; }
1630:           if (file < 0 || file > 7 || rank < 0) return 0;
1631:           out->sq[HESq(file, rank)] = *p;
1632:         file++;
1633:     }
1634: 
1635: 
1636:     while (*p == ' ') p++;
1637:     if (*p) { out->whiteToMove = (*p == 'w'); p++; }
1638: 
1639: 
1640:     while (*p == ' ') p++;
1641:     while (*p && *p != ' ') p++;           // castling rights
1642:     while (*p == ' ') p++;
1643:     if (*p && *p != '-') out->epSquare = HEParseSquare(p);
1644: 
1645: 
1646:     return 1;
1647: }

1661: // Cheapest piece of side `byWhite` attacking sq (0 if none).
1662: static int HECheapestAttacker(const HEBoard *b, int sq, int byWhite) {
1663:     int best = 0;
1664:     int f = HEFile(sq), r = HERank(sq);
1665: #define HE_TAKE(v) do { if (best == 0 || (v) < best) best = (v); } while (0)
1666: 
1667: 
1668:     int pr = byWhite ? r - 1 : r + 1;
1669:     char P = byWhite ? 'P' : 'p';
1670:     if (HEOnBoard(f - 1, pr) && b->sq[HESq(f - 1, pr)] == P) return 100;
1671:     if (HEOnBoard(f + 1, pr) && b->sq[HESq(f + 1, pr)] == P) return 100;
1672: 
1673: 
1674:     char N = byWhite ? 'N' : 'n';
1675:     for (int i = 0; i < 8; i++) {
1676:         int nf = f + HEKnightD[i][0], nr = r + HEKnightD[i][1];
1677:         if (HEOnBoard(nf, nr) && b->sq[HESq(nf, nr)] == N) { HE_TAKE(320); break; }
1678:     }
1679: 
1680: 
1681:     for (int i = 0; i < 4; i++) {
1682:         int nf = f, nr = r;
1683:         for (;;) {
1684:             nf += HEBishopD[i][0]; nr += HEBishopD[i][1];
1685:             if (!HEOnBoard(nf, nr)) break;
1686:             char c = b->sq[HESq(nf, nr)];
1687:             if (c == '.') continue;
1688:             if (c == (byWhite ? 'B' : 'b')) HE_TAKE(330);
1689:             else if (c == (byWhite ? 'Q' : 'q')) HE_TAKE(900);
1690:             break;
1691:         }
1692:     }
1693: 
1694: 
1695:     for (int i = 0; i < 4; i++) {
1696:         int nf = f, nr = r;
1697:         for (;;) {
1698:             nf += HERookD[i][0]; nr += HERookD[i][1];
1699:             if (!HEOnBoard(nf, nr)) break;
1700:             char c = b->sq[HESq(nf, nr)];
1701:             if (c == '.') continue;
1702:             if (c == (byWhite ? 'R' : 'r')) HE_TAKE(500);
1703:             else if (c == (byWhite ? 'Q' : 'q')) HE_TAKE(900);
1704:             break;
1705:         }
1706:     }
1707: 
1708: 
1709:     char K = byWhite ? 'K' : 'k';
1710:     for (int i = 0; i < 8; i++) {
1711:         int nf = f + HEKingD[i][0], nr = r + HEKingD[i][1];
1712:         if (HEOnBoard(nf, nr) && b->sq[HESq(nf, nr)] == K) { HE_TAKE(10000); break; }
1713:     }
1714: #undef HE_TAKE
1715:     return best;
1716: }

1740:     HEMove m = { -1, -1, 0 };

1742:     m.from     = HEParseSquare(uci);
1743:     m.to       = HEParseSquare(uci + 2);

1745:     return m;

1749: static HEBoard HEApply(const HEBoard *in, HEMove m) {
1750:     HEBoard b = *in;
1751:     if (m.from < 0 || m.to < 0) return b;
1752: 
1753: 
1754:     char piece = b.sq[m.from];
1755:     char lower = (char)tolower((unsigned char)piece);
1756:     int     white = HEIsWhitePiece(piece);
1757: 
1758: 
1759:     if (lower == 'p' && m.to == in->epSquare && b.sq[m.to] == '.') {
1760:            int capSq = white ? m.to - 8 : m.to + 8;
1761:            if (capSq >= 0 && capSq < 64) b.sq[capSq] = '.';
1762:     }
1763: 
1764: 
1765:     if (lower == 'k' && abs(HEFile(m.to) - HEFile(m.from)) == 2) {
1766:         int r=HERank(m.from);
1767:         if (HEFile(m.to)==6) {
1768:             b.sq[HESq(5,r)] = b.sq[HESq(7,r)];
1769:             b.sq[HESq(7,r)] = '.';
1770:         } else if (HEFile(m.to)==2) {
1771:             b.sq[HESq(3,r)] = b.sq[HESq(0,r)];
1772:             b.sq[HESq(0,r)] = '.';
1773:         }
1774:     }
1775:     b.sq[m.to] = m.promo ? (white ? (char)toupper((unsigned char)m.promo) : m.promo) : piece;
1776:     b.sq[m.from] = '.';
1777: 
1778:     b.epSquare = -1;
1779:     if (lower == 'p' && abs(HERank(m.to) - HERank(m.from)) == 2)
1780:            b.epSquare = white ? m.from + 8 : m.from - 8;
1781: 
1782: 
1783:     b.whiteToMove = !in->whiteToMove;
1784:     return b;
1785: }

1805: static HEFeatures HEExtract(const HEBoard *b, HEMove m) {
1806:     HEFeatures f;
1807:     memset(&f, 0, sizeof(f));
1808:     if (m.from < 0 || m.to < 0) return f;
1809: 
1810: 
1811:     char piece     = b->sq[m.from];
1812:     char lower     = (char)tolower((unsigned char)piece);
1813:     int    white   = HEIsWhitePiece(piece);
1814:     char target = b->sq[m.to];
1815: 
1816: 
1817:     f.isCapture         = (target != '.') || (lower == 'p' && m.to == b->epSquare);
1818:     f.capturedValue     = (target != '.') ? HEPieceValue(target) : (f.isCapture ? 100 : 0);
1819:     f.isRecapture       = f.isCapture && (m.to == HEOppLastTo);
1820:     f.capturesHigher = f.isCapture && f.capturedValue > HEPieceValue(piece);
1821:     f.isPromotion       = (m.promo != 0);
1822:     f.isPawnPush        = (lower == 'p') && !f.isCapture;
1823:     f.isRookPawn        = (lower == 'p') && (HEFile(m.from) == 0 || HEFile(m.from) == 7);
1824:     f.isCastle          = (lower == 'k') && abs(HEFile(m.to) - HEFile(m.from)) == 2;
1825:     f.isSamePieceAgain = (m.from == HELastOwnTo);
1826: 
1827: 
1828:     int dr = HERank(m.to) - HERank(m.from);
1829:     f.isBackward = white ? (dr < 0) : (dr > 0);
1830: 
1831: 
1832:     int homeRank = white ? 0 : 7;
1833:     f.isDeveloping          = (lower == 'n' || lower == 'b') && (HERank(m.from) == homeRank);
1834:     f.isEarlyQueenSortie = (lower == 'q') && (HERank(m.from) == homeRank);
1835: 
1836: 
1837:     HEBoard after = HEApply(b, m);
1838: 
1839: 
1840:     int oppKing     = HEFindKing(&after, !white);
1841:     f.givesCheck = (oppKing >= 0) && HEAttackedBy(&after, oppKing, white);
1842: 
1843: 
1844:     char oq = white ? 'q' : 'Q';
1845:     for (int i = 0; i < 64; i++) {
1846:           if (after.sq[i] != oq) continue;
1847:           if (HEAttackedBy(&after, i, white)) f.attacksQueen = 1;
1848:           break;
1849:     }
1850: 
1851: 
1852:     f.isQuiet = !f.isCapture && !f.givesCheck && !f.isPromotion && !f.attacksQueen;
1853:     // SEE-lite on the destination square
1854:     int movedValue = HEPieceValue(after.sq[m.to]);
1855:     int cheapest      = HECheapestAttacker(&after, m.to, !white);
1856:     int defended      = HEAttackedBy(&after, m.to, white);
1857:     if (cheapest && (cheapest < movedValue || !defended)) {
1858:         if (f.capturedValue - movedValue < -80) f.hangsPiece = 1;
1859:     }
1860: 
1861: 
1862:     // "Visible" = refutation is one move deep: the moved piece hangs, or any
1863:     // other piece is attacked by something cheaper OR is attacked and undefended.
1864:     f.visibleMistake = f.hangsPiece;
1865:     if (!f.visibleMistake) {
1866:         for (int i = 0; i < 64; i++) {
1867:             char c = after.sq[i];
1868:             if (c == '.' || HEIsWhitePiece(c) != white || i == m.to) continue;
1869:             char l = (char)tolower((unsigned char)c);
1870:             if (l == 'p' || l == 'k') continue;
1871:             int ca = HECheapestAttacker(&after, i, !white);
1872:             if (!ca) continue;
1873:             if (ca < HEPieceValue(c) || !HEAttackedBy(&after, i, white)) {
1874:                    f.visibleMistake = 1;
1875:                    break;
1876:             }
1877:         }
1878:     }
1879:     return f;
1880: }

1917: // How sharp the position is: our attacks, their threats, pressure on their king.
1918: static double HEComplexity(const HEBoard *b, int white) {
1919:     int ours = 0, theirs = 0, nearKing = 0;
1920:     for (int i = 0; i < 64; i++) {
1921:           char c = b->sq[i];
1922:           if (c == '.') continue;
1923:           if (HEIsWhitePiece(c) != white) {
1924:                if (HEAttackedBy(b, i, white)) ours++;
1925:           } else if (tolower((unsigned char)c) != 'k') {
1926:                if (HEAttackedBy(b, i, !white)) theirs++;
1927:           }
1928:     }
1929:     int ok = HEFindKing(b, !white);
1930:     if (ok >= 0) {
1931:           int f = HEFile(ok), r = HERank(ok);
1932:           for (int i = 0; i < 8; i++) {
1933:                int nf = f + HEKingD[i][0], nr = r + HEKingD[i][1];
1934:                if (HEOnBoard(nf, nr) && HEAttackedBy(b, HESq(nf, nr), white)) nearKing++;
1935:           }
1936:     }
1937:     return ours / 6.0 + theirs / 4.0 + nearKing / 4.0;
1938: }
